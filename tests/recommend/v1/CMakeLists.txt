cmake_minimum_required(VERSION 3.20)
project(RecommendationV1Checks LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SHIXU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_subdirectory("${SHIXU_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/shixu")
if(NOT TARGET ShixuRecommendation)
    message(FATAL_ERROR "Recommendation V1 checks require Qt Core/Widgets")
endif()
add_library(ShixuRecommendationV1 STATIC
    "${SHIXU_ROOT}/src/recommendation/RecommendationContractAdapter.cpp")
target_link_libraries(ShixuRecommendationV1 PUBLIC ShixuRecommendation)
add_executable(RecommendationV1Tests test_recommendation_v1.cpp)
target_link_libraries(RecommendationV1Tests PRIVATE ShixuRecommendationV1)
add_test(NAME RecommendationV1Tests COMMAND RecommendationV1Tests)
